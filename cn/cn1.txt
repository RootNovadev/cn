#include <stdio.h>

#define MAX 20

struct node {
    int dist[MAX];
    int from[MAX];
} route[MAX];

int main() {
    int dm[MAX][MAX], n;

    printf("Enter the number of nodes: \n");
    scanf("%d", &n);

    printf("Enter the distance matrix (9999 for infinity): \n");
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            scanf("%d", &dm[i][j]);
            route[i].dist[j] = dm[i][j];
            route[i].from[j] = j;
        }
    }

    int flag;

    do {
        flag = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if (route[i].dist[j] > route[i].dist[k] + route[k].dist[j]) {
                        route[i].dist[j] = route[i].dist[k] + route[k].dist[j];
                        route[i].from[j] = k;
                        flag = 1;
                    }
                }
            }
        }
    } while (flag);

    printf("\nDistance Vector Routing Table:\n");
    printf("Node\tDestination\tNext Hop\tDistance\n");
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            printf("%d\t%d\t\t%d\t\t%d\n", i, j, route[i].from[j], route[i].dist[j]);
        }
    }
    return 0;
}
