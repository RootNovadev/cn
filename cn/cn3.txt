SEVRVER SIDE

#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <stdlib.h>
#include <fcntl.h>
#include <stdio.h>
#include <unistd.h>
#include <string.h>

int main() {
    int s_socket, new_socket;
    int buf_size = 1024;
    char *buf = malloc(buf_size);
    char filename[256];
    struct sockaddr_in server_addr, client_addr;
    socklen_t addr_size;

    s_socket = socket(AF_INET, SOCK_STREAM, 0);
    if (s_socket < 0) {
        perror("Socket creation failed");
        exit(1);
    }

    server_addr.sin_family = AF_INET;
    server_addr.sin_addr.s_addr = INADDR_ANY;
    server_addr.sin_port = htons(6800);

    if (bind(s_socket, (struct sockaddr*)&server_addr, sizeof(server_addr)) < 0) {
        perror("Bind failed");
        exit(1);
    }

    listen(s_socket, 5);
    printf("Server listening on port 6800...\n");

    addr_size = sizeof(client_addr);

    new_socket = accept(s_socket, (struct sockaddr*)&client_addr, &addr_size);
    if (new_socket < 0) {
        perror("Accept failed");
        exit(1);
    }

    recv(new_socket, filename, sizeof(filename), 0);
    printf("Client requested file: %s\n", filename);

    FILE *fp = fopen(filename, "r");
    if (fp == NULL) {
        strcpy(buf, "File not found.\n");
        send(new_socket, buf, strlen(buf), 0);
    } else {
        while (fgets(buf, buf_size, fp) != NULL) {
            send(new_socket, buf, strlen(buf), 0);
        }
        fclose(fp);
    }

    close(new_socket);
    close(s_socket);
    free(buf);

    return 0;
}

#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <stdlib.h>
#include <string.h>
#include <stdio.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int c_socket;
    int buf_size = 1024;
    char *buf = malloc(buf_size);
    char filename;
    struct sockaddr_in address;

    c_socket = socket(AF_INET, SOCK_STREAM, 0);
    if (c_socket < 0) {
        perror("Socket creation failed");
        exit(1);
    }

    address.sin_family = AF_INET;
    address.sin_port = htons(6800);
    inet_pton(AF_INET, "127.0.0.1", &address.sin_addr);

    if (connect(c_socket, (struct sockaddr*)&address, sizeof(address)) < 0) {
        perror("Connection to server failed");
        exit(1);
    }

    printf("Enter file name: ");
    scanf("%s", filename);

    send(c_socket, filename, sizeof(filename), 0);

    int n;
    while ((n = recv(c_socket, buf, buf_size - 1, 0)) > 0) {
        buf[n] = '\0';
        printf("%s", buf);
    }

    close(c_socket);
    free(buf);

    return 0;
}
